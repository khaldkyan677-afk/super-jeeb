const mongoose = require('mongoose');
const jwt = require('jsonwebtoken');
require('dotenv').config();
const User = require('./models/User');
const Store = require('./models/Store');
const Order = require('./models/Order');
const Product = require('./models/Product');

async function runCycle() {
  console.log('🚀 بدء محاكاة دورة حياة التطبيق الكاملة...\n');
  await mongoose.connect(process.env.MONGO_URI);

  // 1. تجهيز الأدمن
  const admin = await User.findOne({ role: 'admin' });
  if (!admin) throw new Error('لا يوجد أدمن في DB');
  const adminToken = jwt.sign({ id: admin._id, role: 'admin' }, process.env.JWT_SECRET);
  const adminHeaders = { 'Authorization': `Bearer ${adminToken}`, 'Content-Type': 'application/json' };

  // 2. إنشاء عميل (Client)
  let client = await User.findOne({ phone: '999999991' });
  if (!client) client = await User.create({ name: 'عميل تجريبي', phone: '999999991', role: 'client' });
  const clientToken = jwt.sign({ id: client._id, role: 'client' }, process.env.JWT_SECRET);
  const clientHeaders = { 'Authorization': `Bearer ${clientToken}`, 'Content-Type': 'application/json' };
  console.log('1️⃣ تم إنشاء عميل تجريبي:', client.name);

  // 3. تقديم طلب متجر (من العميل)
  let store = await Store.findOne({ merchantId: client._id });
  if (!store) {
    store = await Store.create({ name: 'متجر المحاكاة', merchantId: client._id, phone: '999999991', isVerified: false, commission: 5 });
  }
  console.log('2️⃣ العميل قدم طلب فتح متجر (معلق):', store.name);

  // 4. الأدمن يقبل المتجر
  store.isVerified = true;
  store.isActive = true;
  store.documentsVerified = true;
  await store.save();
  console.log('3️⃣ الأدمن قام بتوثيق المتجر (أصبح نشطاً)');

  // 5. التاجر يضيف منتج
  let product = await Product.findOne({ storeId: store._id });
  if (!product) {
    product = await Product.create({ name: 'منتج تجريبي', price: 100, storeId: store._id, category: 'general' });
  }
  console.log('4️⃣ التاجر أضاف منتجاً:', product.name, '| السعر:', product.price);

  // 6. إنشاء مندوب
  let driver = await User.findOne({ phone: '999999992' });
  if (!driver) driver = await User.create({ name: 'مندوب تجريبي', phone: '999999992', role: 'driver' });
  const driverToken = jwt.sign({ id: driver._id, role: 'driver' }, process.env.JWT_SECRET);
  const driverHeaders = { 'Authorization': `Bearer ${driverToken}`, 'Content-Type': 'application/json' };
  console.log('5️⃣ تم إنشاء مندوب تجريبي:', driver.name);

  // 7. العميل يطلب المنتج
  let order = await Order.findOne({ userId: client._id });
  if (!order) {
    order = await Order.create({
      userId: client._id,
      storeId: store._id,
      products: [{ productId: product._id, quantity: 1, price: product.price }],
      total: product.price,
      status: 'pending',
      paymentMethod: 'cash'
    });
  }
  console.log('6️⃣ العميل أنشأ طلباً (قيد الانتظار) | الإجمالي:', order.total);

  // 8. المندوب يقبل الطلب
  order.driverId = driver._id;
  order.status = 'accepted';
  await order.save();
  console.log('7️⃣ المندوب قبل الطلب وذهب للمتجر');

  // 9. المندوب يستلم ويوصل الطلب
  order.status = 'delivered';
  await order.save();
  console.log('8️⃣ المندوب سلم الطلب للعميل (مكتمل)');

  // 10. حساب العمولة والمديونية
  const commissionAmount = (order.total * (store.commission || 5)) / 100;
  const merchantNet = order.total - commissionAmount;
  console.log('9️⃣ العمولة المحسوبة للتطبيق:', commissionAmount, '| صافي حق التاجر:', merchantNet);
  
  // هل يوجد حقل للمديونية في المتجر؟
  if (store.schema.path('walletBalance')) {
    store.walletBalance = (store.walletBalance || 0) + merchantNet;
    await store.save();
    console.log('🔟 تمت إضافة المبلغ لمحفظة التاجر');
  } else {
    console.log('⚠️ تنبيه: لا يوجد حقل Wallet/Balance في نموذج المتجر لحفظ المديونية!');
  }

  // تنظيف البيانات التجريبية (إزالة الطلب والمنتج والمتجر والمندوب والعميل)
  await Order.findByIdAndDelete(order._id);
  await Product.findByIdAndDelete(product._id);
  await Store.findByIdAndDelete(store._id);
  await User.findByIdAndDelete(driver._id);
  await User.findByIdAndDelete(client._id);
  
  console.log('\n🏁 انتهت المحاكاة وتم تنظيف البيانات.');
  await mongoose.disconnect();
}

runCycle().catch(err => console.error('❌ فشل في المحاكاة:', err));

const mongoose = require('mongoose');
require('dotenv').config();
const User = require('./models/User');
const Store = require('./models/Store');
const Product = require('./models/Product');
const Order = require('./models/Order');
const Wallet = require('./models/Wallet');

async function runFinanceSimulation() {
  console.log('🎬 بدء محاكاة الدفع كاش، كود التسليم، والمديونية...\n');
  await mongoose.connect(process.env.MONGO_URI);

  const id = Date.now().toString().slice(-4);
  const pass = '123456';

  // 1. تجهيز المستخدمين (نضع حد أقصى للمديونية 1000 ريال مثلاً)
  const merchant = await User.create({ name: 'التاجر', phone: `771${id}`, role: 'merchant', email: `m${id}@t.com`, password: pass });
  const customer = await User.create({ name: 'الزبون', phone: `772${id}`, role: 'client', email: `c${id}@t.com`, password: pass });
  const driver = await User.create({ name: 'المندوب', phone: `773${id}`, role: 'driver', email: `d${id}@t.com`, password: pass, maxDebtLimit: 1000 }); // حد المديونية 1000

  const store = await Store.create({ name: 'متجر كاش', merchantId: merchant._id, category: 'مطاعم', isVerified: true, isActive: true, commission: 5 });
  const product = await Product.create({ name: 'وجبة كاش', price: 1000, storeId: store._id, isAvailable: true });
  
  const deliveryFee = 500;
  const deliveryPin = '1234'; // كود التسليم الذي سيطلبه المندوب من الزبون

  // 2. الزبون يطلب (الدفع كاش)
  let order = await Order.create({
    customerId: customer._id, merchantId: merchant._id, storeId: store._id,
    items: [{ name: product.name, price: product.price, quantity: 1 }],
    total: product.price + deliveryFee,
    status: 'ready',
    paymentMethod: 'cash', // الدفع كاش
    deliveryPin: deliveryPin,
    isCashCollected: false
  });
  console.log(`🛒 1. الزبون طلب وجبة كاش. الإجمالي المطلوب للمندوب: 1500 ريال.`);

  // 3. المندوب يقبل الطلب
  order.driverId = driver._id;
  order.status = 'delivering';
  await order.save();
  console.log(`🏍️ 2. المندوب قبل الطلب وفي الطريق للزبون.`);

  // 4. المندوب يصل ويطلب الإغلاق (محاولة خاطئة)
  console.log(`🔒 3. المندوب وصل وطلب إغلاق الطلب (بدون كود)...`);
  if (order.deliveryPin !== 'wrong') {
    console.log(`❌ رفض الإغلاق: يجب إدخال كود التسليم الصحيح من الزبون!`);
  }

  // 5. إدخال الكود الصحيح (الكاش تم تحصيله)
  console.log(`🔑 4. المندوب أدخل الكود الصحيح (1234) واستلم الكاش.`);
  order.status = 'delivered';
  order.isCashCollected = true;
  await order.save();

  // 6. الحسابات المالية في حالة الكاش
  const appCommStore = (product.price * 5) / 100; // 50
  const appCommDelivery = (deliveryFee * 15) / 100; // 75
  const totalAppRevenue = appCommStore + appCommDelivery; // 125
  
  const merchantShare = product.price - appCommStore; // 950
  const driverShare = deliveryFee - appCommDelivery; // 425

  // المندوب معه 1500 كاش. من حقه 425. إذن هو مدين للتطبيق بـ (1500 - 425) = 1075
  const driverDebt = order.total - driverShare; // 1075

  console.log(`\n💰 5. التقرير المالي للدفع كاش:`);
  console.log(`   - التطبيق أرباحه: ${totalAppRevenue} ريال.`);
  console.log(`   - التاجر مستحقاته: ${merchantShare} ريال.`);
  console.log(`   - المندوب مستحقاته: ${driverShare} ريال.`);
  console.log(`   - المندوب يحمل كاش: ${order.total} ريال.`);
  
  // تحديث المحافظ
  await Wallet.create({ userId: merchant._id, balance: merchantShare }); // التاجر دائن
  await Wallet.create({ userId: driver._id, balance: -driverDebt }); // المندوب مدين (سالب)
  console.log(`\n💳 6. تم تحديث المحافظ: التاجر (+${merchantShare}) | المندوب (${-driverDebt})`);

  // 7. محاكاة محاولة المندوب استلام طلب جديد (الحد الأقصى للمديونية)
  console.log(`\n⚠️ 7. المندوب حاول استلام طلب جديد...`);
  const driverWallet = await Wallet.findOne({ userId: driver._id });
  const currentDebt = Math.abs(driverWallet.balance); // 1075
  
  if (currentDebt >= driver.maxDebtLimit) {
    console.log(`❌ مرفوض! لقد وصلت للحد الأقصى للمديونية (${driver.maxDebtLimit}). يجب التسديد أولاً.`);
  } else {
    console.log(`✅ يمكنه استلام طلب جديد.`);
  }

  // تنظيف
  await Order.findByIdAndDelete(order._id);
  await Product.findByIdAndDelete(product._id);
  await Store.findByIdAndDelete(store._id);
  await Wallet.deleteMany({ userId: { $in: [merchant._id, driver._id] } });
  await User.deleteMany({ _id: { $in: [merchant._id, customer._id, driver._id] } });

  console.log('\n🏁 انتهت محاكاة الكاش والمديونية.');
  await mongoose.disconnect();
}

runFinanceSimulation().catch(err => console.error('❌ خطأ:', err.message));

function generateSecurePin() {
  let pin;
  let isWeak = true;
  const weak = ['1234','2345','3456','4567','5678','6789','9876','8765','7654','6543','5432','4321','0000','1111','2222','3333','4444','5555','6666','7777','8888','9999'];
  while (isWeak) {
    pin = Math.floor(1000 + Math.random() * 9000).toString();
    if (!weak.includes(pin) && !(pin[0] === pin[2] && pin[1] === pin[3])) isWeak = false;
  }
  return pin;
}

exports.requestWithdrawal = async (req, res) => {
  try {
    const { amount, payoutAccountId } = req.body;
    const userId = req.user._id;

    if (!amount || amount <= 0) return res.status(400).json({ message: 'المبلغ غير صحيح' });

    const acc = await PayoutAccount.findOne({ _id: payoutAccountId, userId });
    if (!acc) return res.status(404).json({ message: 'حساب الاستلام غير موجود' });

    // التحقق من الرصيد
    const wallet = await Wallet.findOne({ userId });
    const balance = wallet?.balance || 0;
    if (balance < amount) {
      return res.status(400).json({ message: 'الرصيد غير كافٍ. المتاح: ' + balance });
    }

    const wd = await Withdrawal.create({
      withdrawalId: genWid(),
      userId,
      amount,
      payoutAccountId,
      status: 'pending'
    });

    await AuditLog.create({
      userId,
      userRole: req.user.role,
      action: 'withdrawal_request',
      entity: 'Withdrawal',
      entityId: wd._id,
      description: `طلب سحب ${amount}`
    });

    res.status(201).json(wd);
  } catch (e) { res.status(500).json({ message: e.message }); }
};

    if (!wd) return res.status(404).json({ message: 'غير موجود' });

    if (wd.status !== 'pending' && status !== 'paid') {
      return res.status(400).json({ message: 'لا يمكن تعديل هذا الطلب' });
    }

    const update = {
      status,
      adminNote,
      processedBy: req.user._id,
      processedAt: new Date()
    };
    if (status === 'paid') {
      update.paidAt = new Date();
      update.transactionRef = transactionRef;
      // خصم من المحفظة
      await Wallet.findOneAndUpdate(
        { userId: wd.userId },
        { $inc: { balance: -wd.amount } }
      );
    }

    const updated = await Withdrawal.findByIdAndUpdate(wd._id, update, { returnDocument: 'after' });

    await AuditLog.create({
      userId: req.user._id,
      userRole: req.user.role,
      action: 'withdrawal_' + status,
      entity: 'Withdrawal',
      entityId: wd._id,
      description: `${status} - ${wd.amount}`
    });

    res.json(updated);
  } catch (e) { res.status(500).json({ message: e.message }); }
};

function gen6() {
  return Math.floor(100000 + Math.random() * 900000).toString();
}

// إرسال كود OTP
exports.sendOtp = async (req, res) => {
  try {
    const { phone, purpose } = req.body;
    if (!phone) return res.status(400).json({ message: 'رقم الجوال مطلوب' });

    // حد أقصى 3 أكواد في 5 دقائق
    const recent = await OtpCode.countDocuments({
      phone,
      createdAt: { $gte: new Date(Date.now() - 5 * 60 * 1000) }
    });
    if (recent >= 3) {
      return res.status(429).json({ message: 'محاولات كثيرة، حاول بعد 5 دقائق' });
    }

    const code = gen6();
    const expiresAt = new Date(Date.now() + 5 * 60 * 1000);

    await OtpCode.create({
      phone,
      code,
      purpose: purpose || 'login',
      expiresAt,
      ip: req.ip
    });

    // TODO: إرسال SMS حقيقي

    res.json({ message: 'تم إرسال الكود', expiresIn: 300 });
  } catch (e) { res.status(500).json({ message: e.message }); }
};

// التحقق من OTP
exports.verifyOtp = async (req, res) => {
  try {
    const { phone, code, purpose } = req.body;
    if (!phone || !code) return res.status(400).json({ message: 'البيانات ناقصة' });

    const otp = await OtpCode.findOne({
      phone,
      purpose: purpose || 'login',
      isUsed: false,
      expiresAt: { $gt: new Date() }
    }).sort({ createdAt: -1 });

    if (!otp) return res.status(400).json({ message: 'الكود غير صحيح أو منتهي' });

    if (otp.attempts >= otp.maxAttempts) {
      return res.status(429).json({ message: 'تجاوزت عدد المحاولات' });
    }

    if (otp.code !== code) {
      await OtpCode.findByIdAndUpdate(otp._id, { attempts: otp.attempts + 1 });
      return res.status(400).json({ message: 'كود خاطئ' });
    }

    await OtpCode.findByIdAndUpdate(otp._id, { isUsed: true });

    await LoginLog.create({
      phone,
      method: 'otp',
      success: true,
      ip: req.ip
    });

    res.json({ message: 'OK', verified: true });
  } catch (e) { res.status(500).json({ message: e.message }); }
};

exports.createPayment = async (req, res) => {
  try {
    const { orderId, method, amount, idempotencyKey } = req.body;
    const userId = req.user._id;

    // Idempotency: لو موجود سابقاً، نرجع نفس النتيجة
    if (idempotencyKey) {
      const existing = await Transaction.findOne({ idempotencyKey });
      if (existing) return res.json(existing);
    }

    const order = await Order.findById(orderId);
    if (!order) return res.status(404).json({ message: 'الطلب غير موجود' });

    const tx = await Transaction.create({
      transactionId: genTxId(),
      userId,
      orderId,
      type: 'payment',
      method: method || 'cash',
      amount: amount || order.total,
      status: 'pending',
      idempotencyKey
    });

    // للكاش: نعتبره مدفوع فوراً
    if (method === 'cash') {
      tx.status = 'success';
      tx.paidAt = new Date();
      await Transaction.findByIdAndUpdate(tx._id, {
        status: 'success',
        paidAt: new Date()
      });
      await Order.findByIdAndUpdate(orderId, {
        paymentMethod: 'cash',
        paymentStatus: 'paid',
        paymentTransactionId: tx._id,
        paidAt: new Date()
      });
      // إنشاء إيصال
      await Receipt.create({
        receiptNumber: 'RC' + Date.now(),
        transactionId: tx._id,
        orderId,
        userId,
        amount: tx.amount,
        method: 'cash'
      });
    }

    res.status(201).json(tx);
  } catch (e) { res.status(500).json({ message: e.message }); }
};

    if (success) {

        status: 'success',
        paidAt: new Date(),
        gatewayRef
      });
      if (tx.orderId) {
        await Order.findByIdAndUpdate(tx.orderId, {
          paymentStatus: 'paid',
          paidAt: new Date(),
          paymentTransactionId: tx._id
        });
      }
    } else {

    }

    res.json({ message: 'OK' });
  } catch (e) { res.status(500).json({ message: e.message }); }

exports.getMyStore = async (req, res) => {
  try {
    const Store = require('../models/Store');
    const Order = require('../models/Order');
    const store = await Store.findOne({ merchantId: req.user._id });
    if (!store) return res.status(404).json({ message: 'Store not found' });

    const today = new Date();
    today.setHours(0, 0, 0, 0);

    const [newCount, preparing, delivered, totalRevenue, todayOrders] = await Promise.all([
      Order.countDocuments({ storeId: store._id, status: 'pending' }),
      Order.countDocuments({ storeId: store._id, status: { $in: ['accepted', 'preparing', 'ready'] } }),
      Order.countDocuments({ storeId: store._id, status: 'delivered' }),
      Order.aggregate([
        { $match: { storeId: store._id, status: 'delivered' } },
        { $group: { _id: null, sum: { $sum: '$total' } } }
      ]),
      Order.countDocuments({ storeId: store._id, createdAt: { $gte: today } }),
    ]);

    res.json({
      store: {
        _id: store._id,
        name: store.name,
        category: store.category,
        logoUrl: store.logoUrl,
        coverUrl: store.coverUrl,
        rating: store.rating,
        isActive: store.isActive,
      },
      stats: {
        newOrders: newCount,
        preparing: preparing,
        delivered: delivered,
        totalRevenue: totalRevenue[0]?.sum || 0,
        todayOrders: todayOrders,
      },
    });
  } catch (e) { res.status(500).json({ message: e.message }); }
};

function generateSecurePin() {
  let pin;
  let isWeak = true;
  const weakPatterns = ['1234','2345','3456','4567','5678','6789','9876','8765','7654','6543','5432','4321','0000','1111','2222','3333','4444','5555','6666','7777','8888','9999'];
  
  while (isWeak) {
    pin = Math.floor(1000 + Math.random() * 9000).toString();
    // نتحقق: ليس في القائمة الضعيفة، وليس به نمط مكرر مثل 1212
    if (!weakPatterns.includes(pin) && !(pin[0] === pin[2] && pin[1] === pin[3])) {
      isWeak = false;
    }
  }
  return pin;
}

const jwt = require('jsonwebtoken');

const protect = async (req, res, next) => {
  let token;
  if (req.headers.authorization && req.headers.authorization.startsWith('Bearer')) {
    try {
      token = req.headers.authorization.split(' ')[1];
      const decoded = jwt.verify(token, process.env.JWT_SECRET);
      req.user = await User.findById(decoded.id).select('-password');
      next();
    } catch (error) {
      return res.status(401).json({ message: 'غير مصرح، التوكن غير صالح' });
    }
  } else {
    return res.status(401).json({ message: 'غير مصرح، لا يوجد توكن' });
  }
};

const authorize = (...roles) => {
  return (req, res, next) => {
    if (!roles.includes(req.user.role)) {
      return res.status(403).json({ message: 'ليس لديك صلاحية للقيام بهذا الإجراء' });
    }
    next();
  };
};

module.exports = { protect, authorize };

const AuditLog = require('../models/AuditLog');
const ErrorLog = require('../models/ErrorLog');

// تسجيل تلقائي للأخطاء
exports.errorLogger = (err, req, res, next) => {
  ErrorLog.create({
    level: 'error',
    message: err.message || 'Unknown error',
    stack: err.stack,
    endpoint: req.originalUrl,
    method: req.method,
    userId: req.user?._id,
    ip: req.ip,
    meta: { body: req.body, query: req.query }
  }).catch(() => {});
  next(err);
};

// تسجيل يدوي لعمليات حساسة
exports.auditAction = (action, entity) => async (req, res, next) => {
  const originalJson = res.json.bind(res);
  res.json = (body) => {
    if (res.statusCode < 400) {
      AuditLog.create({
        userId: req.user?._id,
        userRole: req.user?.role,
        action,
        entity,
        entityId: body?._id,
        ip: req.ip,
        userAgent: req.headers['user-agent'],
        description: `${action} on ${entity}`
      }).catch(() => {});
    }
    return originalJson(body);
  };
  next();
};
